import { Card, CardGrid, LinkCard } from "@astrojs/starlight/components";

```jp title="servers.jp"
~~ Keys before the first header belong to the document root.
version: 2

[SERVER_ID]
config: {
  disabled_channels:,          ~~ empty, on purpose
  disabled_users: [9892, 82082, 8209]
}

[SERVER_ID_2]
prefix: "!"
modules: {
  moderation: true,
  fun: false
}
```

<CardGrid>
  <Card title="Sections on top" icon="list-format">
    A `[SERVER_ID]` header reads better than another brace. Headers may be
    dotted to nest, or quoted when a name contains a dot.
  </Card>
  <Card title="JSON below it" icon="seti:json">
    Inside a section it is objects and arrays, as deep as you like, in the
    shape you already know.
  </Card>
  <Card title="No ceremony" icon="pencil">
    Unquoted keys, `~~` and `~* … *~` comments anywhere, and trailing commas
    quietly ignored.
  </Card>
  <Card title="Empty is a value" icon="approve-check-circle">
    `disabled_channels:,` means the key exists and has nothing in it yet — a
    real state in a config.
  </Card>
</CardGrid>
